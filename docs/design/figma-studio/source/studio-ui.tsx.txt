      viewBox="0 0 24 24"

}
export function Button({
  tone = 'ghost',
  icon,
  className = '',
  children,
  ...props
}: ButtonHTMLAttributes<HTMLButtonElement> & {
  tone?: 'primary' | 'secondary' | 'ghost'
  icon?: string
}) {
  return (
    <button type="button" {...props} className={`studio-button ${tone} ${className}`}>
      {icon && <Icon name={icon} />} {children}
    </button>
  )
}

export function ChromeHeading({ text }: { text: string }) {
  const id = useId().replace(/[^a-zA-Z0-9_-]/g, '')
  const width = Math.max(280, text.length * 36)
  return (
    <h1 className="chrome-heading">
      <span className="sr-only">{text}</span>
      <svg aria-hidden="true" viewBox={`0 0 ${width} 72`} style={{ width: `${width * 0.88}px` }}>
        <defs>
          <linearGradient id={id} x1="0%" x2="0%" y1="0%" y2="100%">
            <stop offset="10%" stopColor="#FFFFFF" />
            <stop offset="43%" stopColor="#C9C9E0" />
            <stop offset="54%" stopColor="#6E6E8F" />
            <stop offset="85%" stopColor="#E6E6F5" />
          </linearGradient>
        </defs>
        <text
          x="1"
          y="56"
          fill={`url(#${id})`}
          textLength={width - 6}
          lengthAdjust="spacingAndGlyphs"
        >
          {text}
        </text>
      </svg>
    </h1>
  )
}
